```jsx highlight={12}
import { useRive } from "@rive-app/react-canvas";

export default function Menu({ onClose }) {
  const { rive, RiveComponent } = useRive({
    src: "/menu.riv",
    stateMachine: "State Machine 1",
    autoplay: true,
    autoBind: true,
  });

  const closeMenu = () => {
    rive?.clearFocus();
    onClose();
  };

  return (
    <>
      <RiveComponent />
      <button onClick={closeMenu}>Close menu</button>
    </>
  );
}
```
